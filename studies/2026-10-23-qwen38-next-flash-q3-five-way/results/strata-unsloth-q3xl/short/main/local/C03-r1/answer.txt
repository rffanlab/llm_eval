def schedule(tasks, capacity):
    if not tasks:
        return {"jobs": [], "makespan": 0}

    ids = set(t['id'] for t in tasks)

    for t in tasks:
        if t['units'] > capacity:
            raise ValueError("units exceeds capacity")
        for d in t['deps']:
            if d not in ids:
                raise ValueError("unknown dependency")

    in_degree = {t['id']: 0 for t in tasks}
    adj = {t['id']: [] for t in tasks}
    for t in tasks:
        for d in t['deps']:
            adj[d].append(t['id'])
            in_degree[t['id']] += 1

    queue = [tid for tid, deg in in_degree.items() if deg == 0]
    count = 0
    while queue:
        node = queue.pop(0)
        count += 1
        for nb in adj[node]:
            in_degree[nb] -= 1
            if in_degree[nb] == 0:
                queue.append(nb)

    if count != len(tasks):
        raise ValueError("circular dependency")

    time = 0
    available = capacity
    running = []
    completed = set()
    started = set()
    jobs = []
    total = len(tasks)

    while len(completed) < total:
        new_running = []
        for (end_time, task) in running:
            if end_time <= time:
                completed.add(task['id'])
                available += task['units']
            else:
                new_running.append((end_time, task))
        running = new_running

        candidates = sorted(
            [t for t in tasks
             if t['id'] not in started
             and all(d in completed for d in t['deps'])],
            key=lambda t: t['id']
        )

        for task in candidates:
            if task['units'] <= available:
                s = time
                e = time + task['duration']
                jobs.append({"id": task['id'], "start": s, "end": e})
                started.add(task['id'])
                running.append((e, task))
                available -= task['units']

        if running:
            time = min(et for et, _ in running)
        elif len(completed) < total:
            raise ValueError("deadlock")

    jobs.sort(key=lambda j: j['id'])
    makespan = max(j['end'] for j in jobs) if jobs else 0

    return {"jobs": jobs, "makespan": makespan}
