def schedule(tasks, capacity):
    if not tasks:
        return {"jobs": [], "makespan": 0}

    task_map = {}
    for t in tasks:
        task_map[t["id"]] = t

    for t in tasks:
        if t["units"] > capacity:
            raise ValueError("units exceeds capacity")

    for t in tasks:
        for d in t["deps"]:
            if d not in task_map:
                raise ValueError("unknown dependency")

    state = {t["id"]: 0 for t in tasks}

    def _dfs(tid):
        if state[tid] == 1:
            raise ValueError("circular dependency")
        if state[tid] == 2:
            return
        state[tid] = 1
        for d in task_map[tid]["deps"]:
            _dfs(d)
        state[tid] = 2

    for t in tasks:
        _dfs(t["id"])

    completed = set()
    started = set()
    running = {}
    available = capacity
    now = 0
    result = {}

    while len(completed) < len(tasks):
        done_now = [rid for rid, end in running.items() if end <= now]
        for rid in done_now:
            completed.add(rid)
            available += task_map[rid]["units"]
            del running[rid]

        ready = []
        for tid in task_map:
            if tid in started:
                continue
            if all(d in completed for d in task_map[tid]["deps"]):
                ready.append(tid)

        ready.sort()

        for tid in ready:
            need = task_map[tid]["units"]
            if available >= need:
                available -= need
                started.add(tid)
                end = now + task_map[tid]["duration"]
                running[tid] = end
                result[tid] = (now, end)

        if running:
            now = min(running.values())
        elif len(completed) < len(tasks):
            raise ValueError("deadlock")

    jobs = sorted(
        [{"id": tid, "start": s, "end": e} for tid, (s, e) in result.items()],
        key=lambda x: x["id"],
    )
    makespan = max(e for _, e in result.values()) if result else 0

    return {"jobs": jobs, "makespan": makespan}
