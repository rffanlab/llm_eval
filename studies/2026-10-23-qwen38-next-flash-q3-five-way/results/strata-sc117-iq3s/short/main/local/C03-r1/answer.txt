def schedule(tasks, capacity):
    if not tasks:
        return {"jobs": [], "makespan": 0}

    task_map = {}
    for t in tasks:
        task_map[t["id"]] = t

    all_ids = set(task_map.keys())

    for t in tasks:
        if t["units"] > capacity:
            raise ValueError("units > capacity")

    for t in tasks:
        for d in t["deps"]:
            if d not in all_ids:
                raise ValueError("unknown dependency")

    in_degree = {tid: 0 for tid in all_ids}
    dependents = {tid: [] for tid in all_ids}
    for t in tasks:
        for d in t["deps"]:
            in_degree[t["id"]] += 1
            dependents[d].append(t["id"])

    queue = [tid for tid in all_ids if in_degree[tid] == 0]
    head = 0
    count = 0
    while head < len(queue):
        tid = queue[head]
        head += 1
        count += 1
        for dep in dependents[tid]:
            in_degree[dep] -= 1
            if in_degree[dep] == 0:
                queue.append(dep)

    if count != len(all_ids):
        raise ValueError("circular dependency")

    completed = set()
    running = {}
    started = set()
    jobs = []
    time = 0
    sorted_ids = sorted(all_ids)

    while True:
        newly_completed = [tid for tid, end in running.items() if end <= time]
        for tid in newly_completed:
            del running[tid]
            completed.add(tid)

        available = capacity - sum(task_map[tid]["units"] for tid in running)

        for tid in sorted_ids:
            if tid in started:
                continue
            t = task_map[tid]
            if not all(d in completed for d in t["deps"]):
                continue
            if t["units"] <= available:
                started.add(tid)
                end_time = time + t["duration"]
                running[tid] = end_time
                jobs.append({"id": tid, "start": time, "end": end_time})
                available -= t["units"]

        if not running:
            break

        time = min(running.values())

    jobs.sort(key=lambda j: j["id"])
    makespan = max((j["end"] for j in jobs), default=0)

    return {"jobs": jobs, "makespan": makespan}
