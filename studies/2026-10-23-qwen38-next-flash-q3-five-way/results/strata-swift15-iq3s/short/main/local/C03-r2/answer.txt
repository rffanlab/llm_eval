def schedule(tasks, capacity):
    if not tasks:
        return {"jobs": [], "makespan": 0}

    task_map = {}
    for t in tasks:
        if t["units"] > capacity:
            raise ValueError("units exceeds capacity")
        task_map[t["id"]] = t

    for t in tasks:
        for d in t["deps"]:
            if d not in task_map:
                raise ValueError("unknown dependency")

    in_degree = {t["id"]: 0 for t in tasks}
    adj = {t["id"]: [] for t in tasks}
    for t in tasks:
        for d in t["deps"]:
            adj[d].append(t["id"])
            in_degree[t["id"]] += 1

    queue = [tid for tid in in_degree if in_degree[tid] == 0]
    count = 0
    while queue:
        node = queue.pop(0)
        count += 1
        for nb in adj[node]:
            in_degree[nb] -= 1
            if in_degree[nb] == 0:
                queue.append(nb)

    if count != len(tasks):
        raise ValueError("circular dependency")

    sorted_ids = sorted(task_map.keys())
    running = []
    completed = set()
    started = set()
    available = capacity
    t = 0
    jobs = []

    while len(completed) < len(tasks):
        new_running = []
        for (end, tid, units) in running:
            if end <= t:
                completed.add(tid)
                available += units
            else:
                new_running.append((end, tid, units))
        running = new_running

        for tid in sorted_ids:
            if tid in started:
                continue
            task = task_map[tid]
            if all(d in completed for d in task["deps"]):
                if task["units"] <= available:
                    started.add(tid)
                    end_time = t + task["duration"]
                    running.append((end_time, tid, task["units"]))
                    available -= task["units"]
                    jobs.append({"id": tid, "start": t, "end": end_time})

        if len(completed) == len(tasks):
            break

        if running:
            t = min(end for (end, _, _) in running)
        else:
            break

    makespan = max(j["end"] for j in jobs) if jobs else 0
    jobs.sort(key=lambda x: x["id"])

    return {"jobs": jobs, "makespan": makespan}
