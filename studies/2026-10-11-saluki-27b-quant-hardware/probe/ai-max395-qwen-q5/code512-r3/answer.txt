"""
A library of 30 small pure functions for validating invoice dictionaries,
sorting events, and summarizing amounts.
"""

import re
from typing import Any, Dict, List, Optional, Tuple, Union


def validate_invoice_dict(invoice: Any) -> bool:
    """
    Validate that the input is a dictionary.
    
    Args:
        invoice: The object to validate.
        
    Returns:
        True if invoice is a dict, False otherwise.
    """
    return isinstance(invoice, dict)


def has_required_keys(invoice: Dict[str, Any], required_keys: List[str]) -> bool:
    """
    Check if the invoice dictionary contains all required keys.
    
    Args:
        invoice: The invoice dictionary.
        required_keys: List of required key names.
        
    Returns:
        True if all required keys are present, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    return all(key in invoice for key in required_keys)


def validate_invoice_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the invoice has a non-empty string 'id' field.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        True if 'id' exists and is a non-empty string, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    invoice_id = invoice.get('id')
    return isinstance(invoice_id, str) and len(invoice_id.strip()) > 0


def validate_invoice_date(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the invoice has a valid date string in 'date' field.
    Expected format: YYYY-MM-DD.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        True if 'date' exists and matches YYYY-MM-DD format, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    date_str = invoice.get('date')
    if not isinstance(date_str, str):
        return False
    pattern = r'^\d{4}-\d{2}-\d{2}$'
    return bool(re.match(pattern, date_str))
