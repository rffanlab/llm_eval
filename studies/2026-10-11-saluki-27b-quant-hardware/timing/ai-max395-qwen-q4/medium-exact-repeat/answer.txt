def validate_invoice_structure(invoice: Dict[str, Any]) -> bool:
    """
    Validate that an invoice dictionary contains the required top-level keys.
    
    Required keys: 'invoice_id', 'customer_id', 'items', 'total_amount', 'currency'.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if all required keys are present, False otherwise.
    """
    required_keys = {'invoice_id', 'customer_id', 'items', 'total_amount', 'currency'}
    return required_keys.issubset(invoice.keys())


def validate_invoice_id(invoice_id: Any) -> bool:
    """
    Validate that the invoice ID is a non-empty string.
    
    Args:
        invoice_id: The invoice ID to validate.
        
    Returns:
        True if valid, False otherwise.
    """
    return isinstance(invoice_id, str) and len(invoice_id.strip()) > 0


def validate_customer_id(customer_id: Any) -> bool:
    """
    Validate that the customer ID is a non-empty string or positive integer.
    
    Args:
        customer_id: The customer ID to validate.
        
    Returns:
        True if valid, False otherwise.
    """
    if isinstance(customer_id, str):
        return len(customer_id.strip()) > 0
    if isinstance(customer_id, int):
        return customer_id > 0
    return False
