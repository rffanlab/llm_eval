All functions are pure: they do not modify their inputs and have no side effects.

def validate_invoice_structure(invoice: Any) -> bool:
    """
    Validate that the input is a dictionary with required top-level keys.
    
    Required keys: 'id', 'amount', 'currency', 'date', 'items'.
    
    Args:
        invoice: The object to validate.
        
    Returns:
        True if the structure is valid, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    required_keys = {'id', 'amount', 'currency', 'date', 'items'}
    return required_keys.issubset(invoice.keys())


def validate_invoice_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the invoice ID is a non-empty string.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        True if the ID is valid, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    inv_id = invoice.get('id')
    return isinstance(inv_id, str) and len(inv_id) > 0


def validate_invoice_amount(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the invoice amount is a positive number (int or float).
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        True if the amount is valid, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    amount = invoice.get('amount')
    if not isinstance(amount, (int, float)):
        return False
    if isinstance(amount, bool):  # bool is subclass of int, exclude it
        return False
    return amount > 0
