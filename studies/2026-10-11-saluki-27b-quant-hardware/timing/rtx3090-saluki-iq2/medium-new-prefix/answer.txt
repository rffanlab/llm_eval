def validate_invoice_structure(invoice: Dict[str, Any]) -> bool:
    """
    Validate that an invoice dictionary contains the required top-level keys.

    Required keys: 'invoice_id', 'customer_id', 'items', 'total_amount'.
    Returns True if all keys are present, False otherwise.
    """
    required_keys = {'invoice_id', 'customer_id', 'items', 'total_amount'}
    return required_keys.issubset(invoice.keys())


def validate_invoice_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'invoice_id' field is a non-empty string.
    Returns True if valid, False otherwise.
    """
    invoice_id = invoice.get('invoice_id')
    return isinstance(invoice_id, str) and len(invoice_id) > 0


def validate_customer_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'customer_id' field is a non-empty string.
    Returns True if valid, False otherwise.
    """
    customer_id = invoice.get('customer_id')
    return isinstance(customer_id, str) and len(customer_id) > 0


def validate_items_list(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'items' field is a non-empty list.
    Returns True if valid, False otherwise.
    """
    items = invoice.get('items')
    return isinstance(items, list) and len(items) > 0
