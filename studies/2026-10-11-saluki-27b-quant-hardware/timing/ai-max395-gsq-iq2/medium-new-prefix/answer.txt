def validate_invoice_keys(invoice: Dict[str, Any]) -> bool:
    """
    Check that an invoice dictionary contains the required top-level keys.

    Required keys: 'id', 'amount', 'currency', 'date', 'vendor'.
    Returns True if all keys are present, False otherwise.
    """
    required = {'id', 'amount', 'currency', 'date', 'vendor'}
    return required.issubset(invoice.keys())


def validate_invoice_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'id' field is a non-empty string.
    Returns True if valid, False otherwise.
    """
    val = invoice.get('id')
    return isinstance(val, str) and len(val) > 0


def validate_invoice_amount(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'amount' field is a positive number (int or float).
    Returns True if valid, False otherwise.
    """
    val = invoice.get('amount')
    return isinstance(val, (int, float)) and val > 0


def validate_invoice_currency(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the 'currency' field is a non-empty string of length 3.
    Returns True if valid, False otherwise.
    """
    val = invoice.get('currency')
    return isinstance(val, str) and len(val) == 3
