"""
Invoice Validation, Event Sorting, and Amount Summarization Library.

This module provides a collection of small, pure functions for:
1. Validating invoice dictionaries.
2. Sorting events based on various criteria.
3. Summarizing financial amounts.

All functions are pure (no side effects) and designed to be easily testable.

# --- Invoice Validation Functions ---

def validate_invoice_structure(invoice: dict) -> bool:
    """
    Validate that an invoice dictionary contains the required top-level keys.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if the structure is valid, False otherwise.
    """
    required_keys = {"invoice_id", "customer_id", "items", "total_amount"}
    return all(key in invoice for key in required_keys)


def validate_invoice_id(invoice: dict) -> bool:
    """
    Validate that the invoice_id is a non-empty string.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if the invoice_id is valid, False otherwise.
    """
    if not validate_invoice_structure(invoice):
        return False
    invoice_id = invoice.get("invoice_id")
    return isinstance(invoice_id, str) and len(invoice_id) > 0


def validate_customer_id(invoice: dict) -> bool:
    """
    Validate that the customer_id is a non-empty string.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if the customer_id is valid, False otherwise.
    """
    if not validate_invoice_structure(invoice):
        return False
    customer_id = invoice.get("customer_id")
    return isinstance(customer_id, str) and len(customer_id) > 0


def validate_items_list(invoice: dict) -> bool:
    """
    Validate that the items field is a non-empty list.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if the items list is valid, False otherwise.
    """
    if not validate_invoice_structure(invoice):
        return False
    items = invoice.get("items")
    return isinstance(items, list) and len(items) > 0
