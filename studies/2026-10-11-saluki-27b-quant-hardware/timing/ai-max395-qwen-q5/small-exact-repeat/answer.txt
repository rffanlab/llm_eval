"""
A library of 30 small pure functions for validating invoice dictionaries,
sorting events, and summarizing amounts.
"""

def validate_invoice_dict(invoice):
    """
    Validate that the input is a dictionary.
    
    Args:
        invoice: The object to validate.
        
    Returns:
        bool: True if invoice is a dict, False otherwise.
    """
    return isinstance(invoice, dict)

def has_required_keys(invoice, keys):
    """
    Check if the invoice dictionary contains all required keys.
    
    Args:
        invoice: The invoice dictionary.
        keys: A list of required key names.
        
    Returns:
        bool: True if all keys are present, False otherwise.
    """
    if not isinstance(invoice, dict):
        return False
    return all(key in invoice for key in keys)

def get_invoice_id(invoice):
    """
    Retrieve the invoice ID from the dictionary.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        str or None: The invoice ID if present, None otherwise.
    """
    if not isinstance(invoice, dict):
        return None
    return invoice.get('id')

def get_invoice_amount(invoice):
    """
    Retrieve the invoice amount from the dictionary.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        float or None: The invoice amount if present and numeric, None otherwise.
    """
    if not isinstance(invoice, dict):
        return None
    amount = invoice.get('amount')
    if isinstance(amount, (int, float)):
        return float(amount)
    return None

def is_amount_positive(invoice):
    """
    Check if the invoice amount is positive.
    
    Args:
        invoice: The invoice dictionary.
        
    Returns:
        bool: True if amount is positive, False otherwise.
    """
    amount = get_invoice_amount(invoice)
    return amount is not None and amount > 0
