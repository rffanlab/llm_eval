def validate_invoice_structure(invoice: Dict[str, Any]) -> bool:
    """
    Validate that an invoice dictionary contains the required top-level keys.
    
    Required keys: 'invoice_id', 'customer_id', 'items', 'total_amount', 'currency'.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if all required keys are present, False otherwise.
    """
    required_keys = {'invoice_id', 'customer_id', 'items', 'total_amount', 'currency'}
    return required_keys.issubset(invoice.keys())


def validate_invoice_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the invoice_id is a non-empty string.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if invoice_id is a non-empty string, False otherwise.
    """
    invoice_id = invoice.get('invoice_id')
    return isinstance(invoice_id, str) and len(invoice_id) > 0


def validate_customer_id(invoice: Dict[str, Any]) -> bool:
    """
    Validate that the customer_id is a non-empty string.
    
    Args:
        invoice: The invoice dictionary to validate.
        
    Returns:
        True if customer_id is a non-empty string, False otherwise.
    """
    customer_id = invoice.get('customer_id')
    return isinstance(customer_id, str) and len(customer_id) > 0
